     1→// ============================================================
     2→// WEBUOS Business Taxonomy Data
     3→// Interconnected hierarchy: Ecosystem → Core Sector → Category
     4→// Each Category has Related Products, Services, and Business Profiles
     5→// ============================================================
     6→
     7→export interface BusinessProfile {
     8→  name: string;
     9→  type: string;
    10→  location: string;
    11→}
    12→
    13→export interface ProductOrService {
    14→  name: string;
    15→  type: "product" | "service";
    16→  description: string;
    17→}
    18→
    19→export interface Category {
    20→  id: string;
    21→  code: string;
    22→  name: string;
    23→  description: string;
    24→  products: ProductOrService[];
    25→  services: ProductOrService[];
    26→  businessProfiles: BusinessProfile[];
    27→}
    28→
    29→export interface CoreSector {
    30→  id: string;
    31→  code: string;
    32→  number: string;
    33→  name: string;
    34→  description: string;
    35→  icon: string;
    36→  color: string;
    37→  categories: Category[];
    38→}
    39→
    40→export interface Ecosystem {
    41→  id: string;
    42→  code: string;
    43→  number: string;
    44→  name: string;
    45→  shortName: string;
    46→  tagline: string;
    47→  description: string;
    48→  icon: string;
    49→  accent: "gold" | "blue" | "purple";
    50→  image: string;
    51→  categories: CoreSector[];
    52→}

   930→export function getEcosystem(id: string): Ecosystem | undefined {
   931→  return ecosystems.find((e) => e.id === id);
   932→}
   933→
   934→export function getCoreSector(
   935→  ecosystemId: string,
   936→  sectorId: string
   937→): { ecosystem: Ecosystem; sector: CoreSector } | undefined {
   938→  const ecosystem = getEcosystem(ecosystemId);
   939→  if (!ecosystem) return undefined;
   940→  const sector = ecosystem.categories.find((c) => c.id === sectorId);
   941→  if (!sector) return undefined;
   942→  return { ecosystem, sector };
   943→}
   944→
   945→export function getCategory(
   946→  ecosystemId: string,
   947→  sectorId: string,
   948→  categoryId: string
   949→): { ecosystem: Ecosystem; sector: CoreSector; category: Category } | undefined {
   950→  const result = getCoreSector(ecosystemId, sectorId);
   951→  if (!result) return undefined;
   952→  const category = result.sector.categories.find((c) => c.id === categoryId);
   953→  if (!category) return undefined;
   954→  return { ecosystem: result.ecosystem, sector: result.sector, category };
   955→}
